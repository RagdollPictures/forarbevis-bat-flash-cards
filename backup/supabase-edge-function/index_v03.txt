import "jsr:@supabase/functions-js/edge-runtime.d.ts";
import { withSupabase } from "jsr:@supabase/server@^1";

const TABLE_NAMES = [
  "final_exams",
  "courses",
  "levels",
  "units",
  "bonus_levels",
  "questions",
  "level_graphics",
  "shared_ui_text",
] as const;

const DELETE_ORDER = [
  "questions",
  "level_graphics",
  "bonus_levels",
  "units",
  "levels",
  "shared_ui_text",
  "final_exams",
  "courses",
] as const;

const BATCH_SIZE = 200;
const READ_BATCH_SIZE = 1000;

type DeleteItem = {
  id: string;
  courseId?: string;
};

function scopedKey(
  courseId: string,
  id: string
) {
  return `${courseId}\u0000${id}`;
}

export default {
  fetch: withSupabase(
    { auth: "none" },
    async (req, ctx) => {
      if (req.method !== "POST") {
        return Response.json(
          {
            error:
              "Method not allowed",
          },
          { status: 405 }
        );
      }

      const expectedSecret =
        Deno.env.get(
          "QUIZ_ENGINE_SYNC_SECRET"
        );

      const providedSecret =
        req.headers.get(
          "x-quiz-engine-secret"
        );

      if (
        !expectedSecret ||
        providedSecret !==
          expectedSecret
      ) {
        return Response.json(
          {
            error:
              "Unauthorized",
          },
          { status: 401 }
        );
      }

      let payload;

      try {
        payload =
          await req.json();
      } catch {
        return Response.json(
          {
            error:
              "Invalid JSON",
          },
          { status: 400 }
        );
      }

      const {
        appId,
        courseIds,
        tables,
        apply = false,
        expectedDeleteCounts,
      } = payload;

      if (
        typeof appId !==
          "string" ||
        !appId.trim()
      ) {
        return Response.json(
          {
            error:
              "Invalid appId",
          },
          { status: 400 }
        );
      }

      if (
        !Array.isArray(
          courseIds
        ) ||
        courseIds.length === 0
      ) {
        return Response.json(
          {
            error:
              "Invalid courseIds",
          },
          { status: 400 }
        );
      }

      const managedCourseIds =
        new Set<string>();

      for (
        const value of
        courseIds
      ) {
        if (
          typeof value !==
            "string" ||
          !value.trim()
        ) {
          return Response.json(
            {
              error:
                "Invalid courseId in courseIds",
              value,
            },
            { status: 400 }
          );
        }

        const courseId =
          value.trim();

        if (
          managedCourseIds.has(
            courseId
          )
        ) {
          return Response.json(
            {
              error:
                `Duplicate courseId "${courseId}"`,
            },
            { status: 400 }
          );
        }

        managedCourseIds.add(
          courseId
        );
      }

      if (
        !tables ||
        typeof tables !==
          "object"
      ) {
        return Response.json(
          {
            error:
              "Invalid tables",
          },
          { status: 400 }
        );
      }

      const sheetKeys:
        Record<
          string,
          Set<string>
        > = {};

      for (
        const tableName of
        TABLE_NAMES
      ) {
        const rows =
          tables[tableName];

        if (
          !Array.isArray(rows)
        ) {
          return Response.json(
            {
              error:
                `Missing or invalid table: ${tableName}`,
            },
            { status: 400 }
          );
        }

        if (
          rows.length === 0
        ) {
          return Response.json(
            {
              error:
                `Table ${tableName} is empty`,
            },
            { status: 400 }
          );
        }

        const keys =
          new Set<string>();

        for (
          const row of rows
        ) {
          if (
            tableName ===
            "shared_ui_text"
          ) {
            if (
              typeof row.key !==
                "string" ||
              !row.key.trim()
            ) {
              return Response.json(
                {
                  error:
                    "Missing key in shared_ui_text",
                },
                { status: 400 }
              );
            }

            const key =
              row.key.trim();

            if (
              keys.has(key)
            ) {
              return Response.json(
                {
                  error:
                    `Duplicate key "${key}" in shared_ui_text`,
                },
                { status: 400 }
              );
            }

            keys.add(key);

            continue;
          }

          if (
            tableName ===
            "courses"
          ) {
            if (
              row.app_id !==
              appId
            ) {
              return Response.json(
                {
                  error:
                    "Wrong app_id in courses",
                  rowId:
                    row.id ??
                    null,
                  receivedAppId:
                    row.app_id ??
                    null,
                  expectedAppId:
                    appId,
                },
                { status: 400 }
              );
            }

            if (
              typeof row.id !==
                "string" ||
              !row.id.trim()
            ) {
              return Response.json(
                {
                  error:
                    "Missing id in courses",
                },
                { status: 400 }
              );
            }

            const id =
              row.id.trim();

            if (
              keys.has(id)
            ) {
              return Response.json(
                {
                  error:
                    `Duplicate id "${id}" in courses`,
                },
                { status: 400 }
              );
            }

            keys.add(id);

            continue;
          }

          if (
            typeof row.course_id !==
              "string" ||
            !managedCourseIds.has(
              row.course_id
            )
          ) {
            return Response.json(
              {
                error:
                  `Invalid course_id in ${tableName}`,
                rowId:
                  row.id ??
                  null,
                receivedCourseId:
                  row.course_id ??
                  null,
                allowedCourseIds:
                  Array.from(
                    managedCourseIds
                  ),
              },
              { status: 400 }
            );
          }

          if (
            typeof row.id !==
              "string" ||
            !row.id.trim()
          ) {
            return Response.json(
              {
                error:
                  `Missing id in ${tableName}`,
              },
              { status: 400 }
            );
          }

          const key =
            scopedKey(
              row.course_id,
              row.id
            );

          if (
            keys.has(key)
          ) {
            return Response.json(
              {
                error:
                  `Duplicate id "${row.id}" for course "${row.course_id}" in ${tableName}`,
              },
              { status: 400 }
            );
          }

          keys.add(key);
        }

        sheetKeys[
          tableName
        ] = keys;
      }

      const courseSheetIds =
        sheetKeys.courses;

      if (
        courseSheetIds.size !==
        managedCourseIds.size
      ) {
        return Response.json(
          {
            error:
              "courseIds does not match courses table",
            courseIds:
              Array.from(
                managedCourseIds
              ),
            courses:
              Array.from(
                courseSheetIds
              ),
          },
          { status: 400 }
        );
      }

      for (
        const courseId of
        managedCourseIds
      ) {
        if (
          !courseSheetIds.has(
            courseId
          )
        ) {
          return Response.json(
            {
              error:
                `courseId "${courseId}" is missing from courses table`,
            },
            { status: 400 }
          );
        }
      }

      async function readAllRows(
        tableName: string,
        scopeCourseIds:
          string[]
      ) {
        const allRows:
          Record<
            string,
            unknown
          >[] = [];

        let start = 0;

        while (true) {
          let response;

          if (
            tableName ===
            "shared_ui_text"
          ) {
            response =
              await ctx.supabaseAdmin
                .from(
                  "shared_ui_text"
                )
                .select("key")
                .order(
                  "key",
                  {
                    ascending:
                      true,
                  }
                )
                .range(
                  start,
                  start +
                    READ_BATCH_SIZE -
                    1
                );
          } else if (
            tableName ===
            "courses"
          ) {
            response =
              await ctx.supabaseAdmin
                .from("courses")
                .select("id")
                .eq(
                  "app_id",
                  appId
                )
                .order(
                  "id",
                  {
                    ascending:
                      true,
                  }
                )
                .range(
                  start,
                  start +
                    READ_BATCH_SIZE -
                    1
                );
          } else {
            response =
              await ctx.supabaseAdmin
                .from(tableName)
                .select(
                  "course_id,id"
                )
                .in(
                  "course_id",
                  scopeCourseIds
                )
                .order(
                  "course_id",
                  {
                    ascending:
                      true,
                  }
                )
                .order(
                  "id",
                  {
                    ascending:
                      true,
                  }
                )
                .range(
                  start,
                  start +
                    READ_BATCH_SIZE -
                    1
                );
          }

          if (
            response.error
          ) {
            throw new Error(
              response.error.message
            );
          }

          const rows =
            response.data ?? [];

          allRows.push(
            ...rows
          );

          if (
            rows.length <
            READ_BATCH_SIZE
          ) {
            break;
          }

          start +=
            READ_BATCH_SIZE;
        }

        return allRows;
      }

      let existingCourseRows;

      try {
        existingCourseRows =
          await readAllRows(
            "courses",
            []
          );
      } catch (error) {
        return Response.json(
          {
            error:
              "Could not read courses",
            details:
              error instanceof
              Error
                ? error.message
                : String(
                    error
                  ),
          },
          { status: 500 }
        );
      }

      const existingAppCourseIds =
        existingCourseRows
          .map(
            (row) =>
              typeof row.id ===
              "string"
                ? row.id
                : null
          )
          .filter(
            (
              value
            ): value is string =>
              value !== null
          );

      const scopedCourseIds =
        Array.from(
          new Set([
            ...existingAppCourseIds,
            ...Array.from(
              managedCourseIds
            ),
          ])
        );

      const result: Record<
        string,
        {
          sheet: number;
          database: number;
          new: number;
          existing: number;
          wouldDelete: number;
        }
      > = {};

      const deleteItems:
        Record<
          string,
          DeleteItem[]
        > = {};

      for (
        const tableName of
        TABLE_NAMES
      ) {
        const rows =
          tables[tableName];

        let databaseRows;

        try {
          if (
            tableName ===
            "courses"
          ) {
            databaseRows =
              existingCourseRows;
          } else {
            databaseRows =
              await readAllRows(
                tableName,
                scopedCourseIds
              );
          }
        } catch (error) {
          return Response.json(
            {
              error:
                `Could not read ${tableName}`,
              details:
                error instanceof
                Error
                  ? error.message
                  : String(
                      error
                    ),
            },
            { status: 500 }
          );
        }

        const databaseKeys =
          new Set<string>();

        const databaseItems =
          new Map<
            string,
            DeleteItem
          >();

        for (
          const row of
          databaseRows
        ) {
          if (
            tableName ===
            "shared_ui_text"
          ) {
            if (
              typeof row.key !==
              "string"
            ) {
              continue;
            }

            databaseKeys.add(
              row.key
            );

            databaseItems.set(
              row.key,
              {
                id: row.key,
              }
            );

            continue;
          }

          if (
            tableName ===
            "courses"
          ) {
            if (
              typeof row.id !==
              "string"
            ) {
              continue;
            }

            databaseKeys.add(
              row.id
            );

            databaseItems.set(
              row.id,
              {
                id: row.id,
              }
            );

            continue;
          }

          if (
            typeof row.id !==
              "string" ||
            typeof row.course_id !==
              "string"
          ) {
            continue;
          }

          const key =
            scopedKey(
              row.course_id,
              row.id
            );

          databaseKeys.add(
            key
          );

          databaseItems.set(
            key,
            {
              id: row.id,
              courseId:
                row.course_id,
            }
          );
        }

        const currentSheetKeys =
          sheetKeys[
            tableName
          ];

        let newCount = 0;
        let existingCount = 0;

        for (
          const key of
          currentSheetKeys
        ) {
          if (
            databaseKeys.has(
              key
            )
          ) {
            existingCount++;
          } else {
            newCount++;
          }
        }

        const itemsToDelete:
          DeleteItem[] = [];

        if (
          tableName !==
          "shared_ui_text"
        ) {
          for (
            const key of
            databaseKeys
          ) {
            if (
              !currentSheetKeys.has(
                key
              )
            ) {
              const item =
                databaseItems.get(
                  key
                );

              if (item) {
                itemsToDelete.push(
                  item
                );
              }
            }
          }
        }

        deleteItems[
          tableName
        ] =
          itemsToDelete;

        result[
          tableName
        ] = {
          sheet:
            rows.length,
          database:
            databaseKeys.size,
          new:
            newCount,
          existing:
            existingCount,
          wouldDelete:
            itemsToDelete.length,
        };
      }

      if (apply !== true) {
        return Response.json({
          ok: true,
          dryRun: true,
          appId,
          courseIds:
            Array.from(
              managedCourseIds
            ),
          tables:
            result,
        });
      }

      const totalDeletes =
        TABLE_NAMES.reduce(
          (
            sum,
            tableName
          ) =>
            sum +
            result[
              tableName
            ].wouldDelete,
          0
        );

      if (
        totalDeletes > 0
      ) {
        if (
          !expectedDeleteCounts ||
          typeof expectedDeleteCounts !==
            "object"
        ) {
          return Response.json(
            {
              error:
                "Delete confirmation required",
              tables:
                result,
            },
            { status: 409 }
          );
        }

        for (
          const tableName of
          TABLE_NAMES
        ) {
          const expected =
            expectedDeleteCounts[
              tableName
            ];

          const actual =
            result[
              tableName
            ].wouldDelete;

          if (
            expected !== actual
          ) {
            return Response.json(
              {
                error:
                  "Delete counts changed",
                table:
                  tableName,
                expected,
                actual,
              },
              { status: 409 }
            );
          }
        }
      }

      for (
        const tableName of
        TABLE_NAMES
      ) {
        const rows =
          tables[tableName];

        for (
          let start = 0;
          start <
          rows.length;
          start +=
            BATCH_SIZE
        ) {
          const batch =
            rows.slice(
              start,
              start +
                BATCH_SIZE
            );

          let error;

          if (
            tableName ===
            "shared_ui_text"
          ) {
            const response =
              await ctx.supabaseAdmin
                .from(
                  "shared_ui_text"
                )
                .upsert(
                  batch,
                  {
                    onConflict:
                      "key",
                  }
                );

            error =
              response.error;
          } else if (
            tableName ===
            "courses"
          ) {
            const response =
              await ctx.supabaseAdmin
                .from(
                  "courses"
                )
                .upsert(
                  batch,
                  {
                    onConflict:
                      "id",
                  }
                );

            error =
              response.error;
          } else {
            const response =
              await ctx.supabaseAdmin
                .from(
                  tableName
                )
                .upsert(
                  batch,
                  {
                    onConflict:
                      "course_id,id",
                  }
                );

            error =
              response.error;
          }

          if (error) {
            console.error(
              error
            );

            return Response.json(
              {
                error:
                  `Upsert failed for ${tableName}`,
                details:
                  error.message,
                batchStart:
                  start,
              },
              { status: 500 }
            );
          }
        }
      }

      for (
        const tableName of
        DELETE_ORDER
      ) {
        const items =
          deleteItems[
            tableName
          ] ?? [];

        if (
          tableName ===
          "shared_ui_text"
        ) {
          continue;
        }

        if (
          tableName ===
          "courses"
        ) {
          const ids =
            items.map(
              (item) =>
                item.id
            );

          for (
            let start = 0;
            start <
            ids.length;
            start +=
              BATCH_SIZE
          ) {
            const batch =
              ids.slice(
                start,
                start +
                  BATCH_SIZE
              );

            const response =
              await ctx.supabaseAdmin
                .from(
                  "courses"
                )
                .delete()
                .eq(
                  "app_id",
                  appId
                )
                .in(
                  "id",
                  batch
                );

            if (
              response.error
            ) {
              console.error(
                response.error
              );

              return Response.json(
                {
                  error:
                    "Delete failed for courses",
                  details:
                    response
                      .error
                      .message,
                  batchStart:
                    start,
                },
                {
                  status: 500,
                }
              );
            }
          }

          continue;
        }

        const grouped =
          new Map<
            string,
            string[]
          >();

        for (
          const item of
          items
        ) {
          if (
            !item.courseId
          ) {
            continue;
          }

          const current =
            grouped.get(
              item.courseId
            ) ?? [];

          current.push(
            item.id
          );

          grouped.set(
            item.courseId,
            current
          );
        }

        for (
          const [
            courseId,
            ids,
          ] of grouped
        ) {
          for (
            let start = 0;
            start <
            ids.length;
            start +=
              BATCH_SIZE
          ) {
            const batch =
              ids.slice(
                start,
                start +
                  BATCH_SIZE
              );

            const response =
              await ctx.supabaseAdmin
                .from(
                  tableName
                )
                .delete()
                .eq(
                  "course_id",
                  courseId
                )
                .in(
                  "id",
                  batch
                );

            if (
              response.error
            ) {
              console.error(
                response.error
              );

              return Response.json(
                {
                  error:
                    `Delete failed for ${tableName}`,
                  details:
                    response
                      .error
                      .message,
                  courseId,
                  batchStart:
                    start,
                },
                {
                  status: 500,
                }
              );
            }
          }
        }
      }

      return Response.json({
        ok: true,
        dryRun: false,
        appId,
        courseIds:
          Array.from(
            managedCourseIds
          ),
        tables:
          result,
        message:
          "Full multi-course sync completed.",
      });
    }
  ),
};
